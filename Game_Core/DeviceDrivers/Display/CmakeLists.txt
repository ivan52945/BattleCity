cmake_minimum_required(VERSION 3.22)

SET(TARGET_LIB display)

# Display is a header-only module now: an abstract display interface
# (Inc/disp_iface.hpp). There is no compiled adapter - the platform
# layer (main.cpp) creates the concrete driver and passes it into Game.
add_library(${TARGET_LIB} INTERFACE)

target_include_directories(${TARGET_LIB} INTERFACE Inc)
